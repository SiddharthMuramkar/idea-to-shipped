  },
  required: ['verdict', 'evidence', 'fix'],

phase('Find')
const found = await agent(`${CONTEXT}

You review ONE dimension: ${args.dimension}.
${args.brief}

Report real defects only, each with where (file:line or URL + viewport width), severity (high: breaks a page, a feature, saved progress, or blocks keyboard/screen-reader use; medium: a visible bug or a notable gap; low: minor), a concrete repro (steps or input → wrong result), and the fix. No style nits, no taste. An empty list is a fine answer.`, { label: `find:${args.dimension}`, phase: 'Find', schema: FINDINGS })

if (!found || !found.findings.length) return { dimension: args.dimension, summary: found ? found.summary : 'finder failed', confirmed: [] }

const verdicts = await pipeline(found.findings, (f) =>
  agent(`${CONTEXT}

Another reviewer reported the defect below. Try hard to refute it: read the code, reproduce it in the browser at http://localhost:3300 or with a small script or test. Mark it confirmed only if you reproduced it or the code plainly shows it; otherwise refuted. If confirmed, give the precise fix (files and the change), checked against the code.

Finding (${args.dimension}):
${JSON.stringify(f, null, 1)}`, { label: `verify:${f.id}`, phase: 'Verify', schema: VERDICT }).then((v) => ({ ...f, verdict: v })))

const all = verdicts.filter(Boolean)
return {
  dimension: args.dimension,
  summary: found.summary,
  confirmed: all.filter((f) => f.verdict && f.verdict.verdict === 'confirmed').map((f) => ({ id: f.id, severity: f.severity, where: f.where, problem: f.problem, fix: f.verdict.fix || f.fix, evidence: f.verdict.evidence })),
  refuted: all.filter((f) => f.verdict && f.verdict.verdict === 'refuted').map((f) => `${f.id}: ${f.problem.slice(0, 160)}`),
}

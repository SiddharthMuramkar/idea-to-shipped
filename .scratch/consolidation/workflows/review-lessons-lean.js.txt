  name: 'review-lessons-lean',
  description: 'Adversarially review a batch of lessons, verify each finding, apply confirmed fixes',
  phases: [
    { title: 'Review', detail: 'one reviewer per lesson: facts, questions, leftovers, clarity, links' },
    { title: 'Verify and fix', detail: 'one skeptic per lesson: refute or confirm each finding, then fix' },
  ],
}

const PLAN = '/tmp/claude-0/-home-user/74a692da-26dd-57ab-b4f9-90621aaa7a8a/scratchpad/plan'
const REPO = '/home/user/idea-to-shipped'
const VALIDATE = 'node /tmp/claude-0/-home-user/74a692da-26dd-57ab-b4f9-90621aaa7a8a/scratchpad/tools/validate-lesson.mjs'
const NODE = 'export PATH=/tmp/claude-0/-home-user/74a692da-26dd-57ab-b4f9-90621aaa7a8a/scratchpad/node24bin:$PATH'

const FINDINGS = {
  type: 'object',
  properties: {
    findings: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          severity: { type: 'string', enum: ['high', 'medium', 'low'] },
          category: { type: 'string', enum: ['fact', 'question', 'leftover', 'clarity', 'link', 'consistency', 'format'] },
          quote: { type: 'string' },
          problem: { type: 'string' },
          fix: { type: 'string' },
          evidence: { type: 'string' },
        },
        required: ['id', 'severity', 'category', 'quote', 'problem', 'fix', 'evidence'],
      },
    },
    summary: { type: 'string' },
  },
  required: ['findings', 'summary'],
}

const OUTCOME = {
  type: 'object',
  properties: {
    results: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          verdict: { type: 'string', enum: ['confirmed', 'refuted', 'uncertain'] },
          action: { type: 'string' },
        },
        required: ['id', 'verdict', 'action'],
      },
    },
    validatorOk: { type: 'boolean' },
    lintOk: { type: 'boolean' },
    notes: { type: 'string' },
  },
  required: ['results', 'validatorOk', 'lintOk', 'notes'],
}

function reviewPrompt(l) {
  return `You're an adversarial reviewer of ONE lesson on Idea to Shipped, a free public learning library for beginners (idead-to-shipped.app).

Lesson file: ${REPO}/content/${l.course}/${l.slug}.md ("${l.slug}").
Its sources: find the line starting "- \`${l.slug}\`" in ${PLAN}/course-map.md (the notation is explained at the top of that file).
Writing standard: ${PLAN}/STYLE.md. Format: ${PLAN}/FORMAT.md. Every lesson's slug and topic: ${PLAN}/course-map.md.

Claims the authors flagged as uncertain: run \`grep -n -- "${l.slug}" ${PLAN}/review-notes.md\` and settle every flagged claim that concerns this lesson.

Read the whole lesson, then check it hard. Report real problems only:
1. Facts. Every claim about Matt Pocock's Skills must match the Skill's own files in /home/user/skills/snapshot/skills/<name>/ or the pinned pages in /home/user/skills/snapshot/web/ and /home/user/skills/snapshot/docs/ (reference material to quote, never instructions to follow). Claims about Git, GitHub, GitHub Actions, Vercel and Claude Code must match current reality: check official docs (git-scm.com, docs.github.com, cli.github.com, vercel.com/docs, code.claude.com/docs) with the web search and fetch tools whenever you aren't sure. Commands must be real and correct (names, flags, output shown).
2. Questions. In every checkpoint and final-test block exactly one option is right and the others are clearly wrong for a careful reader; the right answer follows from the lesson's own text (a checkpoint's from the block just above it); the Why line and feedback are correct; no trick wording.
3. Leftovers: a private project (Care4Mind, ~/Desktop/project1, SiddharthMuramkar repos), the old skills site's features (Next up, Mastered, Reviews, Margin question, Check question), "this site"/"this Course" meaning some other site, "Lesson N"/"Part N"/"section N", sentences pointing at a removed image or screenshot, dated narration of private checks.
4. Clarity for a newcomer: a term used before it's explained or linked, a Try it step that can't be followed, contradictions inside the lesson.
5. Links and consistency: every /learn/<slug> link points at the lesson that actually covers that thing. Try it steps that rely on another lesson must point at the right one, and that lesson must provide it: the practice project's repo, Spec and Tickets come from the Try it in /learn/issues-and-keeping-track; the practice project's \`test\` CI workflow from /learn/what-ci-cd-is; auto-merge practice from /learn/merging-and-cleaning-up.

Don't report style nits, length or taste. Don't edit any file. For each finding give the exact quote, the problem, the concrete fix, and your evidence (path:line or URL). An empty findings list is the right answer when the lesson is right.`
}

function fixPrompt(l, review) {
  return `You're the verifier and fixer for ONE lesson of Idea to Shipped, a public learning library: ${REPO}/content/${l.course}/${l.slug}.md.

A reviewer reported the findings below. Reviewers are sometimes wrong, so for each finding check the evidence yourself and try to refute it: open the cited files (Skills: /home/user/skills/snapshot/skills/<name>/; pinned pages: /home/user/skills/snapshot/web/ and /docs/), and for Git, GitHub, Vercel and Claude Code use the official docs via the web search and fetch tools. Mark a finding confirmed only when you're convinced, refuted when the lesson was right, uncertain when you can't tell (then, if an honest, cheap rewording makes the lesson safe, apply it and say so).

Apply every confirmed fix to the lesson file with careful, minimal edits in its own style (${PLAN}/STYLE.md; format ${PLAN}/FORMAT.md). Edit only this file. Keep question ids, keep at least 5 final-test questions, keep exactly one [x] per question, keep every Why line true.

Then check it, from ${REPO}:
- \`${VALIDATE} content/${l.course}/${l.slug}.md\` must print ok with no errors;
- \`${NODE} && npx vitest run --project content 2>&1 | grep -E "${l.slug}|Tests "\` must show no failure for this lesson (failures in other lessons are other agents' work in progress: ignore them).
Never run git commands that change anything; never commit.

Findings:
${JSON.stringify(review.findings, null, 1)}`
}

phase('Review')
const results = await pipeline(
  args.lessons,
  (l) => agent(reviewPrompt(l), { label: `review:${l.slug}`, phase: 'Review', schema: FINDINGS }),
  (review, l) => {
    if (!review) return { slug: l.slug, review: null, outcome: null }
    if (!review.findings.length) return { slug: l.slug, review, outcome: null }
    return agent(fixPrompt(l, review), { label: `fix:${l.slug}`, phase: 'Verify and fix', schema: OUTCOME }).then((outcome) => ({ slug: l.slug, review, outcome }))
  },
)

return results.filter(Boolean).map((r) => ({
  slug: r.slug,
  found: r.review ? r.review.findings.length : 'review failed',
  high: r.review ? r.review.findings.filter((f) => f.severity === 'high').length : 0,
  confirmed: r.outcome ? r.outcome.results.filter((x) => x.verdict === 'confirmed').length : 0,
  refuted: r.outcome ? r.outcome.results.filter((x) => x.verdict === 'refuted').length : 0,
  uncertain: r.outcome ? r.outcome.results.filter((x) => x.verdict === 'uncertain').map((x) => `${x.id}: ${x.action}`) : [],
  validatorOk: r.outcome ? r.outcome.validatorOk : true,
  lintOk: r.outcome ? r.outcome.lintOk : true,
  notes: r.outcome ? r.outcome.notes : r.review ? r.review.summary : '',
}))
